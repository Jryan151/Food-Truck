<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no">
    <title>Digital Storefront</title>

<script>
    // ==========================================
    // 1. INJECTED: UI TOGGLES FOR DEMO
    // ==========================================
    window.toggleSoldOut = function(btn, index) {
        if(event) event.stopPropagation();
        const card = btn.closest('.relative');
        const title = card.querySelector('h3');
        if (btn.innerText.includes('Sold Out')) {
            btn.innerText = 'Simulate Sold Out';
            btn.className = 'bg-zinc-800 hover:bg-zinc-700 text-zinc-300 border border-zinc-700 px-3 py-1.5 rounded-lg text-xs font-bold z-50 ml-3 transition-colors';
            card.classList.remove('opacity-50', 'grayscale');
            if(title) title.classList.remove('line-through', 'text-zinc-500');
        } else {
            btn.innerText = 'Sold Out';
            btn.className = 'bg-red-500/20 text-red-400 border border-red-500/50 px-3 py-1.5 rounded-lg text-xs font-bold z-50 ml-3 transition-colors';
            card.classList.add('opacity-50', 'grayscale');
            if(title) title.classList.add('line-through', 'text-zinc-500');
        }
    };

    window.toggleRushModeDemo = function() {
        const rushEl = document.getElementById('rush-status-container');
        if (!rushEl) return;
        if (rushEl.classList.contains('hidden')) {
            rushEl.classList.remove('hidden'); rushEl.classList.add('flex');
            const waitEl = document.getElementById('cartEstimatedWait');
            if(waitEl) { waitEl.innerText = 'Est. Prep Time: ~45 mins'; waitEl.classList.remove('hidden'); waitEl.classList.add('block'); }
        } else {
            rushEl.classList.add('hidden'); rushEl.classList.remove('flex');
            const waitEl = document.getElementById('cartEstimatedWait');
            if(waitEl) { waitEl.innerText = 'Est. Prep Time: ~15 mins'; waitEl.classList.add('hidden'); waitEl.classList.remove('block'); }
        }
    };

    // ==========================================
    // 2. CORE APPLICATION LOGIC
    // ==========================================
    const app = {
        menu: [
            { name: "Signature Street Taco", price: 3.50, desc: "Our famous hand-pressed tortilla with house ingredients.", soldout: false }
        ],
        cart: [],
        currentModalIndex: -1
    };

    function switchView(view) {
        const landing = document.getElementById('landing-view');
        const menu = document.getElementById('menu-view');
        const tracker = document.getElementById('tracker-view');
        
        if (landing) landing.classList.add('hidden');
        if (menu) menu.classList.add('hidden');
        if (tracker) tracker.classList.add('hidden');
        
        if (view === 'menu') renderMenu();
        const target = document.getElementById(`${view}-view`);
        if (target) target.classList.remove('hidden');
    }

    function renderMenu() {
        const container = document.getElementById('menuList');
        if (!container) return;
        
        const isDemo = sessionStorage.getItem('demoVendorData') !== null;

        container.innerHTML = app.menu.map((item, index) => `
            <div class="relative bg-zinc-900/50 border border-zinc-800 rounded-2xl p-4 flex justify-between items-center cursor-pointer hover:border-zinc-700 transition-colors" onclick="openItemModal(${index})">
                <div class="flex-1 pr-4 text-left">
                    <h3 class="font-bold text-lg item-name text-white">${item.name}</h3>
                    <p class="text-emerald-400 font-bold mt-1">$${item.price.toFixed(2)}</p>
                </div>
                ${isDemo ? `<button class="bg-zinc-800 hover:bg-zinc-700 text-zinc-300 border border-zinc-700 px-3 py-1.5 rounded-lg text-xs font-bold z-50 ml-3 transition-colors" onclick="window.toggleSoldOut(this, ${index});">Simulate Sold Out</button>` : ''}
            </div>
        `).join('');
    }

    function openItemModal(index) {
        app.currentModalIndex = index;
        const item = app.menu[index];
        const nameEl = document.getElementById('modalItemName');
        const descEl = document.getElementById('modalItemDesc');
        const qtyEl = document.getElementById('modalQty');
        const totalEl = document.getElementById('modalTotal');
        
        if (nameEl) nameEl.innerText = item.name;
        if (descEl) descEl.innerText = item.desc || '';
        if (qtyEl) qtyEl.innerText = '1';
        if (totalEl) totalEl.innerText = `$${item.price.toFixed(2)}`;
        
        const modal = document.getElementById('item-modal');
        const content = document.getElementById('itemModalContent');
        if (modal) modal.classList.remove('hidden');
        if (content) setTimeout(() => content.classList.remove('translate-y-full'), 10);
    }

    function adjustModalQty(delta) {
        const qtyEl = document.getElementById('modalQty');
        if (!qtyEl || app.currentModalIndex === -1) return;
        let qty = parseInt(qtyEl.innerText) + delta;
        if (qty < 1) qty = 1;
        qtyEl.innerText = qty;
        const total = qty * app.menu[app.currentModalIndex].price;
        const totalEl = document.getElementById('modalTotal');
        if (totalEl) totalEl.innerText = `$${total.toFixed(2)}`;
    }

    function confirmAddToCart() {
        const qtyEl = document.getElementById('modalQty');
        if (!qtyEl || app.currentModalIndex === -1) return;
        const qty = parseInt(qtyEl.innerText);
        const item = app.menu[app.currentModalIndex];
        app.cart.push({ ...item, qty });
        updateCartBadge();
        closeAllModals();
    }

    function updateCartBadge() {
        const badge = document.getElementById('cartBadge');
        if (!badge) return;
        if (app.cart.length > 0) {
            badge.innerText = app.cart.reduce((sum, item) => sum + item.qty, 0);
            badge.classList.remove('hidden');
        } else {
            badge.classList.add('hidden');
        }
    }

    function openCartModal() {
        renderCart();
        const modal = document.getElementById('cart-modal');
        if (modal) modal.classList.remove('hidden');
    }

    function closeCartModal() {
        const modal = document.getElementById('cart-modal');
        if (modal) modal.classList.add('hidden');
    }

    function renderCart() {
        const list = document.getElementById('cartItemsList');
        if (!list) return;
        let total = 0;
        
        if (app.cart.length === 0) {
            list.innerHTML = `<div class="text-zinc-500 text-center py-4">Your cart is empty</div>`;
        } else {
            list.innerHTML = app.cart.map((item, index) => {
                const itemTotal = item.price * item.qty;
                total += itemTotal;
                return `
                <div class="flex justify-between items-center bg-zinc-950/50 p-3 rounded-xl border border-zinc-800">
                    <div>
                        <div class="font-bold text-sm text-white">${item.name}</div>
                        <div class="text-zinc-500 text-xs mt-1">$${item.price.toFixed(2)} each</div>
                    </div>
                    <div class="flex items-center gap-3">
                        <span class="font-bold text-emerald-400">$${itemTotal.toFixed(2)}</span>
                        <div class="flex items-center bg-zinc-900 rounded-lg">
                            <button onclick="app.cart.splice(${index}, 1); renderCart(); updateCartBadge();" class="px-2 py-1 text-zinc-400 hover:text-red-400 text-xs font-bold">&times;</button>
                        </div>
                    </div>
                </div>`;
            }).join('');
        }
        
        const subtotalEl = document.getElementById('cartSubtotal');
        const totalEl = document.getElementById('cartTotal');
        if (subtotalEl) subtotalEl.innerText = `$${total.toFixed(2)}`;
        if (totalEl) totalEl.innerText = `$${total.toFixed(2)}`;
    }

    function closeAllModals() {
        const content = document.getElementById('itemModalContent');
        if (content) content.classList.add('translate-y-full');
        setTimeout(() => {
            const itemM = document.getElementById('item-modal');
            const cartM = document.getElementById('cart-modal');
            const successM = document.getElementById('success-modal');
            if (itemM) itemM.classList.add('hidden');
            if (cartM) cartM.classList.add('hidden');
            if (successM) successM.classList.add('hidden');
        }, 200);
    }

    function submitOrder() {
        if (app.cart.length === 0) return;
        closeCartModal();
        const successM = document.getElementById('success-modal');
        if (successM) successM.classList.remove('hidden');
        app.cart = [];
        updateCartBadge();
        const trackerBtn = document.getElementById('btn-view-tracker');
        if (trackerBtn) trackerBtn.classList.remove('hidden');
    }

    // ==========================================
    // 3. DEMO DATA CATCHER
    // ==========================================
    window.addEventListener('DOMContentLoaded', () => {
        setTimeout(() => {
            const demoDataStr = sessionStorage.getItem('demoVendorData');
            if (demoDataStr) {
                try {
                    const demo = JSON.parse(demoDataStr);
                    const controls = document.getElementById('demo-controls-panel');
                    if (controls) {
                        controls.classList.remove('hidden');
                        controls.classList.add('flex');
                    }
                    
                    document.querySelectorAll('h1, .vendor-name').forEach(el => {
                        if (el.innerText.includes('Bobbo') || el.innerText.includes('Store Front')) {
                            el.innerText = demo.vendorName || demo.truckName || demo.name || 'My Food Truck';
                        }
                    });

                    if (demo.themeClass === 'theme-silver' || demo.theme === 'Sterling Silver') {
                        document.body.style.setProperty('background-color', '#E5E7EB', 'important');
                        document.querySelectorAll('h1, h2, h3, p, span:not(.text-emerald-400):not(.text-amber-400):not(.text-red-400)').forEach(el => {
                            if (!el.closest('#demo-controls-panel')) el.style.setProperty('color', '#1A1A1A', 'important');
                        });
                        document.querySelectorAll('.bg-zinc-950, .bg-zinc-900').forEach(el => {
                            if (!el.closest('#demo-controls-panel')) {
                                el.style.setProperty('background-color', '#ffffff', 'important');
                                el.style.setProperty('border-color', '#d1d5db', 'important');
                            }
                        });
                    }

                    if (demo.menu && demo.menu.length > 0) {
                        app.menu[0].name = demo.menu[0].name;
                        if (demo.menu[0].price) app.menu[0].price = parseFloat(demo.menu[0].price);
                    }
                } catch (err) {
                    console.error("Demo sync failed:", err);
                }
            }
        }, 150);
    });
</script>
</body>
</html>
